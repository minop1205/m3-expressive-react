# Dialog

MD3 basic dialog: focus is trapped inside, <kbd>Escape</kbd> and scrim
click close it, and focus returns to the opener on close.

export function DialogDemo() {
  const [open, setOpen] = useState(false)
  return (
    <>
      <Button variant="filled" onClick={() => setOpen(true)}>
        Open dialog
      </Button>
      <Dialog
        open={open}
        onClose={() => setOpen(false)}
        title="Reset settings?"
        actions={
          <>
            <Button variant="text" onClick={() => setOpen(false)}>Cancel</Button>
            <Button variant="text" onClick={() => setOpen(false)}>Reset</Button>
          </>
        }
      >
        This restores the default configuration. Your data is kept.
      </Dialog>
    </>
  )
}

centers the headline under it.

## Full-screen

`fullScreen` covers the viewport with a top bar — a close button, the
`title`, and `actions` at the end — over full-bleed content. It has no
scrim; the close button and <kbd>Escape</kbd> call `onClose`. Use it on
compact screens for tasks with several inputs.

export function FullScreenDemo() {
  const [open, setOpen] = useState(false)
  return (
    <>
      <Button variant="outlined" onClick={() => setOpen(true)}>
        Open full-screen dialog
      </Button>
      {/* Raised above the docs navbar; the dialog itself is position: fixed. */}
      <div style={{ position: 'relative', zIndex: 'var(--ifm-z-index-overlay)' }}>
        <Dialog
          open={open}
          fullScreen
          onClose={() => setOpen(false)}
          title="New event"
          actions={<Button variant="text" onClick={() => setOpen(false)}>Save</Button>}
        >
          <div style={{ display: 'flex', flexDirection: 'column', gap: 16, paddingTop: 8 }}>
            <TextField variant="outlined" label="Title" />
            <TextField variant="outlined" label="Location" />
            <TextField variant="outlined" multiline label="Notes" />
          </div>
        </Dialog>
      </div>
    </>
  )
}
