import { useState } from 'react'
import Demo from '@site/src/components/Demo'
import PropsTable from '@site/src/components/PropsTable'
import { TextField, IconButton } from 'm3-expressive-react'
import { SearchIcon, MailIcon } from '@site/src/components/icons/ActionIcons'
import ErrorIcon from '@material-symbols/svg-400/outlined/error.svg'
import VisibilityIcon from '@material-symbols/svg-400/outlined/visibility.svg'
import VisibilityOffIcon from '@material-symbols/svg-400/outlined/visibility_off.svg'

export function PasswordField({ variant }) {
  const [visible, setVisible] = useState(false)
  const id = `password-${variant}`
  return (
    <TextField
      id={id}
      variant={variant}
      label="Password"
      type={visible ? 'text' : 'password'}
      autoComplete="current-password"
      endIcon={
        <IconButton
          variant="standard"
          toggle
          selected={visible}
          onChange={(_event, next) => setVisible(next)}
          icon={<VisibilityIcon />}
          selectedIcon={<VisibilityOffIcon />}
          aria-label="Show password"
          aria-controls={id}
        />
      }
    />
  )
}

# TextField

MD3 text field in filled and outlined variants, with floating label,
supporting text, character counter, and multiline mode.

## Variants

<Demo>
  <TextField variant="filled" label="Filled" />
  <TextField variant="outlined" label="Outlined" />
</Demo>

```tsx
<TextField
  variant="outlined"
  label="Email"
  value={email}
  onChange={(event) => setEmail(event.target.value)}
/>
```

## Tonal style (experimental)

`tonal` applies the Expressive tonal look from Compose
(`TextFieldDefaults.tonalColors()` with `roundedShape`): a 12dp rounded
container and a label that stays on-surface-variant on focus.

- `filled` — a surface-container fill with **no** active indicator line
  (error-container in error).
- `outlined` — a transparent container with an outline-variant border (error
  in error; it thickens to 2dp on focus). The label stays **inside** the
  container, as on `filled`, so the border has no notch.

<Demo>
  <TextField tonal label="Filled tonal" />
  <TextField tonal variant="outlined" label="Outlined tonal" />
  <TextField tonal label="Error" error errorText="Error message" defaultValue="Invalid" />
</Demo>

```tsx
<TextField tonal label="Name" />
<TextField tonal variant="outlined" label="Name" />
```

:::caution Experimental

Compose still marks these values `TODO(b/448727879)`: they are waiting for
real design tokens. The tonal look may change in a minor release once the
spec settles. Because the filled tonal field has no indicator line, focus is
shown mainly by the text cursor; check that this is enough for your design.

:::

## Supporting text & error

<Demo>
  <TextField label="Username" supportingText="Visible to everyone" />
  <TextField label="Password" error supportingText="Too short" />
</Demo>

`error` switches the field to the error colors and sets `aria-invalid`;
`errorText` replaces `supportingText` while in error and is announced
(`role="alert"`).

<Demo>
  <TextField
    label="Email"
    defaultValue="ada@example"
    error
    errorText="Enter a valid email address"
    supportingText="We never share your email"
    endIcon={<ErrorIcon />}
  />
</Demo>

```tsx
<TextField
  label="Email"
  value={email}
  onChange={(event) => setEmail(event.target.value)}
  error={!isValid}
  errorText="Enter a valid email address"
  supportingText="We never share your email"
/>
```

## Label position

By default the label floats: it rests in the input line and minimizes on
focus or once there is a value (the outlined variant notches its border).
`labelPlacement="above"` puts the label above the container instead, always
minimized and with no notch; the placeholder is then visible right away.

<Demo>
  <TextField labelPlacement="above" label="Filled" placeholder="Placeholder" />
  <TextField variant="outlined" labelPlacement="above" label="Outlined" placeholder="Placeholder" />
</Demo>

```tsx
<TextField variant="outlined" labelPlacement="above" label="Email" placeholder="you@example.com" />
```

`shrinkLabel` keeps the floating label minimized even while the field is
empty and unfocused, so the placeholder and prefix / suffix text show
alongside it.

<Demo>
  <TextField shrinkLabel label="Filled" placeholder="Placeholder" />
  <TextField variant="outlined" shrinkLabel label="Amount" prefixText="$" />
</Demo>

```tsx
<TextField shrinkLabel label="Search" placeholder="Try “jacket”" />
```

## Password (secure) field

There is no separate secure field component. Compose's `SecureTextField` is a
text field that hides its value and has a visibility toggle in the trailing
icon slot, and you can build the same thing from a `TextField` and an
`IconButton`:

- `type="password"` hides the value. The browser and the on-screen keyboard
  then handle masking, briefly showing the last character typed on mobile,
  and turning off autocorrect. Set `autoComplete` to `"current-password"` or
  `"new-password"` so password managers can help.
- The toggle is a standard `IconButton` in toggle mode. Its name stays
  **"Show password"** and `aria-pressed` reports whether the password is
  shown, so do not swap the label as well. `aria-controls` points at the
  input.
- Clicking the toggle does not move focus into the input, and the
  `IconButton` keeps its own 48dp touch target inside the 24dp icon slot.

<Demo>
  <PasswordField variant="filled" />
  <PasswordField variant="outlined" />
</Demo>

```tsx
const [visible, setVisible] = useState(false)

<TextField
  id="password"
  label="Password"
  type={visible ? 'text' : 'password'}
  autoComplete="current-password"
  endIcon={
    <IconButton
      variant="standard"
      toggle
      selected={visible}
      onChange={(_event, next) => setVisible(next)}
      icon={<VisibilityIcon />}
      selectedIcon={<VisibilityOffIcon />}
      aria-label="Show password"
      aria-controls="password"
    />
  }
/>
```

## Prefix & suffix text

`prefixText` / `suffixText` show static text around the value, such as a
currency symbol or a unit.

<Demo>
  <TextField label="Price" prefixText="$" defaultValue="24.99" />
  <TextField variant="outlined" label="Weight" suffixText="kg" defaultValue="72" />
</Demo>

```tsx
<TextField label="Price" prefixText="$" />
<TextField variant="outlined" label="Weight" suffixText="kg" />
```

## Character counter

`maxLength` limits the input natively and shows a `length / maxLength`
counter at the end of the supporting line.

<Demo>
  <TextField label="Display name" maxLength={20} supportingText="Shown on your profile" />
</Demo>

```tsx
<TextField label="Display name" maxLength={20} supportingText="Shown on your profile" />
```

## Multiline

`multiline` renders a `<textarea>` that grows with its content; `rows` sets
the initial number of visible rows (2 by default).

<Demo>
  <TextField multiline label="Message" />
  <TextField variant="outlined" multiline rows={4} label="Description" maxLength={200} />
</Demo>

```tsx
<TextField multiline label="Message" />
<TextField variant="outlined" multiline rows={4} label="Description" maxLength={200} />
```

## States

`required` appends `*` to the label and marks the input required;
`disabled`, `readOnly` and `placeholder` are the native input attributes,
styled per MD3. The placeholder shows once the field is focused.

<Demo>
  <TextField label="Full name" required />
  <TextField label="Account ID" readOnly defaultValue="ACC-20931" />
  <TextField label="Disabled" disabled defaultValue="Not editable" />
  <TextField variant="outlined" label="City" placeholder="e.g. Tokyo" />
</Demo>

```tsx
<TextField label="Full name" required />
<TextField label="Account ID" readOnly value={accountId} />
<TextField label="Disabled" disabled value="Not editable" />
<TextField variant="outlined" label="City" placeholder="e.g. Tokyo" />
```

## The native input

`ref` points at the root element; use `inputRef` for the native control,
and `inputProps` for attributes without a dedicated prop:

```tsx
<TextField
  label="Amount"
  inputRef={inputRef}
  inputProps={{ inputMode: 'decimal', pattern: '[0-9]*' }}
/>
```
