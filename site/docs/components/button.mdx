## Icons

`startIcon` / `endIcon` add a decorative leading or trailing icon, sized and
spaced per the button's `size`.

## Toggle buttons

`toggle` makes the button selectable (`aria-pressed`). Control it with
`selected` + `onChange(event, selected)`, or leave it uncontrolled with
`defaultSelected`. Selecting a toggle button morphs its resting shape
(round → square, square → round) and switches it to the selected colors —
an MD3 Expressive behavior. The `text` variant has no toggle styling.

export function ToggleDemo() {
  const [selected, setSelected] = useState(false)
  return (
    <>
      <Button
        variant="tonal"
        toggle
        selected={selected}
        onChange={(event, next) => setSelected(next)}
        startIcon={selected ? <FavoriteIcon /> : <FavoriteBorderIcon />}
      >
        {selected ? 'Saved' : 'Save'}
      </Button>
      <Button variant="outlined" toggle defaultSelected>
        Uncontrolled
      </Button>
      <Button variant="filled" shape="square" toggle>
        Square
      </Button>
    </>
  )
}

</Demo>
